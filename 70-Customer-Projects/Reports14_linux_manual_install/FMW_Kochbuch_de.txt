SCHRITT 1: FMW-Infrastruktur installieren  (als oracle)

  ----------------------------------------------------------------------------
  1.1 Voraussetzungen pruefen
  ----------------------------------------------------------------------------
  # JDK vorhanden, ORACLE_HOME leer, /etc/oraInst.loc vorhanden (aus
  # OS_Kochbuch_de.txt Schritt 5)

  $JDK_HOME/bin/java -version
  # Erwartet: java version "21..." (Oracle, NICHT OpenJDK)

  ls "$ORACLE_HOME" 2>/dev/null && echo "WARNUNG: ORACLE_HOME nicht leer!"
  # Erwartet: keine Ausgabe. "WARNUNG" -> STOP, Inhalt klaeren.

  # Erwartet: inventory_loc=/u01/app/oraInventory, inst_group=oinstall

  ls -l $PATCH_STORAGE/wls/fmw_14.1.2.0.0_infrastructure.jar

  # (Silent) ist die uebliche Wahl auf einem Server ohne dauerhafte
  # grafische Oberflaeche, Option B (Grafisch) fuer alle, die dem
  # Installationsassistenten lieber beim Arbeiten zusehen/Werte interaktiv
  # pruefen wollen. Nur EINE der beiden Optionen ausfuehren:
  #   Option A -> 1.2, danach 1.4
  #   Option B -> 1.3, danach 1.4

  ----------------------------------------------------------------------------
  1.2 Option A: Silent  (Antwortdatei, empfohlen fuer Server ohne GUI)
  ----------------------------------------------------------------------------
  # Response-Datei anlegen. DECLINE_SECURITY_UPDATES=true, da ohne
  # MOS-Zugangsdaten keine automatischen Update-Benachrichtigungen
  # moeglich/gewuenscht sind. ORACLE_HOME in der Datei = $ORACLE_HOME.

  cat > $PATCH_STORAGE/wls/wls_install.rsp << 'EOF'
[ENGINE]
Response File Version=1.0.0.0.0

[GENERIC]
ORACLE_HOME=/u01/oracle/fmw
INSTALL_TYPE=Fusion Middleware Infrastructure
MYORACLESUPPORT_USERNAME=
MYORACLESUPPORT_PASSWORD=
DECLINE_SECURITY_UPDATES=true
SECURITY_UPDATES_VIA_MYORACLESUPPORT=false
PROXY_HOST=
PROXY_PORT=
PROXY_USER=
PROXY_PWD=
COLLECTOR_SUPPORTHUB_URL=
EOF

  grep '^ORACLE_HOME' $PATCH_STORAGE/wls/wls_install.rsp
  # Muss derselbe Pfad sein wie: echo $ORACLE_HOME

  # CV_ASSUME_DISTID=OEL8: Oracle Linux 9 wird vom Installer-internen
  # Systemvoraussetzungs-Check (CVU) noch nicht nativ erkannt. Diese
  # Variable taeuscht NUR die Betriebssystem-Identifikation vor - alle
  # anderen Pruefungen (RAM, Platz, Pakete) laufen normal und ungefiltert.
  # -ignoreSysPrereqs wird bewusst NICHT verwendet.

  export CV_ASSUME_DISTID=OEL8

  $JDK_HOME/bin/java -jar $PATCH_STORAGE/wls/fmw_14.1.2.0.0_infrastructure.jar \
      -silent \
      -responseFile $PATCH_STORAGE/wls/wls_install.rsp \
      -invPtrLoc /etc/oraInst.loc \
      -jreLoc $JDK_HOME
  # Installationslog: $INV_LOC/logs/ (siehe 1.4)

  unset CV_ASSUME_DISTID

  rm -f $PATCH_STORAGE/wls/wls_install.rsp

  # Weiter mit 1.4.

  ----------------------------------------------------------------------------
  1.3 Option B: Grafisch  (Installationsassistent, OUI-GUI)
  ----------------------------------------------------------------------------
  # Voraussetzung: eine grafische Oberflaeche muss erreichbar sein - entweder
  # per X11-Weiterleitung ueber SSH (lokaler X-Server noetig, z.B. Xming/
  # VcXsrv unter Windows, XQuartz unter macOS, ein Linux-Desktop sonst)
  # oder per VNC-Sitzung auf dem Server selbst.
  #
  #   ssh -X oracle@<hostname>          # X11-Weiterleitung aktivieren
  #   echo $DISPLAY                     # muss gesetzt sein, z.B. localhost:10.0
  #
  # Falls "xauth" auf dem Server fehlt, bricht der Installer mit
  # "No protocol specified" oder gar nicht sichtbar ab.
  # Als root, einmalig, falls noch nicht vorhanden:

  dnf install -y xorg-x11-xauth

  # Wieder als oracle (SCHRITT 0 in dieser Sitzung):

  echo $DISPLAY
  # Erwartet: z.B. localhost:10.0 - LEER -> ssh -X fehlt

  export CV_ASSUME_DISTID=OEL8

  # Gleicher Installer-Aufruf wie Option A, aber OHNE -silent/-responseFile -
  # startet stattdessen den grafischen Oracle Universal Installer (OUI):

  $JDK_HOME/bin/java -jar $PATCH_STORAGE/wls/fmw_14.1.2.0.0_infrastructure.jar \
      -invPtrLoc /etc/oraInst.loc \
      -jreLoc $JDK_HOME

  # Im Assistenten dieselben Werte eintragen, die in Option A in der
  # Response-Datei standen:
  #   Oracle Home:         /u01/oracle/fmw  (= $ORACLE_HOME)
  #   Installationstyp:    Fusion Middleware Infrastructure
  #   Security Updates:    "I wish to receive security updates..." abwaehlen
  #                        (keine MOS-Zugangsdaten hinterlegt)
  #   Proxy-Einstellungen: leer lassen, falls kein Proxy benoetigt wird
  #
  # Der Assistent fuehrt am Ende dieselbe Pruefung/Installation aus wie
  # Option A - Dauer und Ergebnis sind identisch.

  unset CV_ASSUME_DISTID

  ----------------------------------------------------------------------------
  1.4 Kontrolle  (beide Optionen)
  ----------------------------------------------------------------------------
  # Dauer (beide Optionen): ca. 10-15 Minuten. Installationslog liegt unter

  # Erwartet: "OPatch succeeded." und die installierten Produkte

  # Die Schleife fragt nach JEDEM prereq: "j" = einspielen, alles andere

  for PATCHNR in 40050062 40035984 40024749 39995927 40057999 39940047 39979746 39726399; do

      cd /tmp/mw_patches/$PATCHNR

SCHRITT 4: SSL-Zertifikat auf Nginx ausrollen  (als root)
================================================================================

  # Voraussetzung: Zertifikat + Schluessel liegen bereits vor - siehe
  # SSL_Kochbuch_de.txt (Option A/B/C/D) fuer die Erzeugung.

  ----------------------------------------------------------------------------
  4.1 Variablen  (als root - aus patch_install.ini ablesen)
  ----------------------------------------------------------------------------
  export SSL_CERT_FILE=/u01/oracle/certs/server.crt

  export SSL_KEY_FILE=/u01/oracle/certs/server.key

  export SSL_CHAIN_FILE=/u01/oracle/certs/chain.crt
  # optional, leer lassen falls nicht vorhanden

  export WLS_SERVER_FQDN=wls01.company.local
  # fuer den Live-Test in 4.4

  ls -l "$SSL_CERT_FILE" "$SSL_KEY_FILE" "$SSL_CHAIN_FILE"
  # Erwartet: Zertifikat und Schluessel vorhanden (Chain nur bei B/C/D)

  ----------------------------------------------------------------------------
  4.2 Zertifikat validieren
  ----------------------------------------------------------------------------
  openssl x509 -noout -enddate -in "$SSL_CERT_FILE"

  openssl x509 -noout -subject -in "$SSL_CERT_FILE"

  openssl x509 -noout -issuer -in "$SSL_CERT_FILE"

  openssl x509 -noout -checkend 0 -in "$SSL_CERT_FILE"
  # Erwartet: Certificate will not expire

  openssl x509 -noout -modulus -in "$SSL_CERT_FILE" | md5sum

  openssl rsa -noout -modulus -in "$SSL_KEY_FILE" | md5sum
  # Beide md5sum-Werte MUESSEN identisch sein (Schluessel passt zum
  # Zertifikat). Abweichend -> STOP, siehe BEKANNTE PROBLEME.

  ----------------------------------------------------------------------------
  4.3 Sichern und ausrollen
  ----------------------------------------------------------------------------
  # Backup des bisherigen Standes (siehe SSL_Kochbuch_de.txt - keine
  # Automatik vorhanden)

  SSL_BAK=/etc/nginx/ssl.bak-$(date +%Y%m%d_%H%M%S)

  cp -a /etc/nginx/ssl $SSL_BAK 2>/dev/null

  ls -ld $SSL_BAK
  # Erwartet: die Sicherung (beim allerersten Ausrollen ggf. leer)

  # Ausrollen: Chain nur anhaengen, wenn vorhanden (Option B/C/D), bei
  # Option A (selbstsigniert) existiert keine Chain-Datei

  mkdir -p /etc/nginx/ssl

  chmod 700 /etc/nginx/ssl

  if [ -s "$SSL_CHAIN_FILE" ]; then
      cat "$SSL_CERT_FILE" "$SSL_CHAIN_FILE" > /etc/nginx/ssl/fullchain.pem
  else
      cp "$SSL_CERT_FILE" /etc/nginx/ssl/fullchain.pem
  fi

  cp "$SSL_KEY_FILE" /etc/nginx/ssl/privkey.pem

  chmod 644 /etc/nginx/ssl/fullchain.pem

  chmod 600 /etc/nginx/ssl/privkey.pem

  chown root:root /etc/nginx/ssl/fullchain.pem /etc/nginx/ssl/privkey.pem

  ls -l /etc/nginx/ssl
  # Erwartet: fullchain.pem -rw-r--r-- root, privkey.pem -rw------- root

  ----------------------------------------------------------------------------
  4.4 Konfiguration testen und Nginx (erstmalig) starten
  ----------------------------------------------------------------------------
  nginx -t
  # Erwartet: "syntax is ok" und "test is successful"

  systemctl start nginx

  systemctl status nginx
  # Erwartet: active (running)

  # Live-Test

  openssl s_client -connect 127.0.0.1:443 -servername "$WLS_SERVER_FQDN" -brief

  curl -o /dev/null -s -w "%{http_code}\n" http://127.0.0.1/

BEKANNTE PROBLEME UND LOESUNGEN
================================================================================

  Problem: Installer bricht mit einem Systemvoraussetzungs-Fehler zur
           Betriebssystem-Erkennung ab
  Ursache: CV_ASSUME_DISTID nicht gesetzt (OL9 wird vom Installer nicht
           nativ erkannt)
  Loesung: export CV_ASSUME_DISTID=OEL8 vor dem Installer-Aufruf setzen
           (Schritt 1.2 / 1.3)

  Problem: OPatch-Upgrade schlaegt mit einem Konflikt zu Patch 23335292 ab
  Ursache: Dieser alte Patch muss laut OPatch-Readme VOR dem Upgrade
           zurueckgerollt werden
  Loesung: Schritt 3.2 in der angegebenen Reihenfolge durchfuehren
           (Rollback -> 30 s warten -> Upgrade)

  Problem: "opatch prereq CheckConflictAgainstOHWithDetail" meldet einen
           Konflikt
  Ursache: Ein anderer, bereits installierter Patch ueberschneidet sich
  Loesung: Betroffenen alten Patch zuerst zurueckrollen, oder beim
           Hersteller nach einem gemeinsamen Merge-Patch fragen - NICHT
           einfach erzwingen

  Problem: "opatch napply" fuer das SPB schlaegt fehl oder haengt
  Ursache: JAVA_HOME/ORACLE_HOME nicht wie in Schritt 3.3 unsetten, oder
           zu wenig freier Plattenplatz (SPB braucht ca. 4 GB ausserhalb
           von $ORACLE_HOME zum Entpacken)
  Loesung: Schritt 3.3 exakt in der gezeigten Reihenfolge wiederholen,
           df -h gegenpruefen

  Problem: Zertifikat und Schluessel liefern unterschiedliche
           md5sum-Werte
  Ursache: Falsches Schluessel/Zertifikat-Paar (z.B. alter Schluessel
           versehentlich wiederverwendet)
  Loesung: Beide Dateien aus demselben Erzeugungsvorgang (siehe
           SSL_Kochbuch_de.txt) neu ausrollen (Schritt 4)

  Problem: curl auf /reports/ liefert 502/503 nach dem SSL-Rollout
  Ursache: WebLogic/Reports-Managed-Server laeuft noch nicht
  Loesung: Normal an dieser Stelle - weiter mit Reports_kochbuch_de.txt
